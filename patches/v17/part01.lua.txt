local function notify(title, text, duration)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = tostring(title),
            Text = tostring(text or ""),
            Duration = duration or 4,
        })
    end)
end

local function tracebackError(err)
    if debug and type(debug.traceback) == "function" then
        return debug.traceback(tostring(err), 2)
    end
    return tostring(err)
end

local function getMapName()
    local map = Workspace:FindFirstChild("Map")
    local trueName = map and map:FindFirstChild("TrueName")
    return trueName and tostring(trueName.Value)
        or (map and map.Name or "Unknown")
end

local function getCTDModule()
    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local moduleScript = modules and modules:FindFirstChild("CTDModule")
    if not moduleScript then
        return nil
    end

    local ok, module = pcall(require, moduleScript)
    return ok and module or nil
end

function session.auto.normalizeDeck(deck)
    local normalized = {}

    for _, unitName in ipairs(type(deck) == "table" and deck or {}) do
        unitName = tostring(unitName or "")
        if unitName ~= "" and unitName ~= "None" then
            normalized[#normalized + 1] = unitName
        end
    end

    return normalized
end

function session.auto.deckExactMatches(a, b)
    a = session.auto.normalizeDeck(a)
    b = session.auto.normalizeDeck(b)

    if #a ~= #b then
        return false
    end

    for i = 1, #a do
        if a[i] ~= b[i] then
            return false
        end
    end

    return true
end

function session.auto.loadoutDataDeck(data)
    local towers = type(data) == "table" and data.Towers or nil
    if type(towers) ~= "table" then
        return {}
    end

    local slots = {}
    for key, value in pairs(towers) do
        local index = tonumber(tostring(key):match("^Tower(%d+)$"))
        if index then
            slots[#slots + 1] = {
                index = index,
                value = tostring(value),
            }
        end
    end

    table.sort(slots, function(a, b)
        return a.index < b.index
    end)

    local deck = {}
    for _, slot in ipairs(slots) do
        deck[#deck + 1] = slot.value
    end

    return session.auto.normalizeDeck(deck)
end

function session.auto.verifyCurrentMacroDeck(macro)
    local required = session.auto.requiredUnitsFromMacro(macro)
    local wanted = session.auto.normalizeDeck(
        type(macro) == "table" and macro.deck or nil
    )
    local current = session.auto.normalizeDeck(session.getDeckSnapshot())

    if #wanted > 0 then
        if session.auto.deckExactMatches(current, wanted) then
            return true
        end

        return false, "expected: "
            .. session.compactDeckText(wanted)
            .. " | current: "
            .. session.compactDeckText(current)
    end

    if #required == 0 then
        return true
    end

    if #current == 0 then
        return false, "PlayerData.Loadout is empty/unavailable"
    end

    local ok, missing = session.auto.deckContainsRequired(current, required)
    if ok then
        return true
    end

    return false, "missing "
        .. tostring(missing)
        .. " | current: "
        .. session.compactDeckText(current)
end

        pcall(module.waitforload)
