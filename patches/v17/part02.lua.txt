            ready = session.auto.deckExactMatches(deck, wantedDeck)
        else
            ready = session.auto.deckContainsRequired(deck, required) == true
        end

        if ready then
            notify(
                "AutoExecute",
                "Deck equipped: " .. session.compactDeckText(deck),
                3
            )
            return true
        end

        task.wait(0.1)
    end

    return false, "deck did not update after EquipLoadout | current: "
        .. session.compactDeckText(session.getDeckSnapshot())
end

function session.auto.isElevatorRoomCandidate(object)
    if not object
        or not (object:IsA("Folder") or object:IsA("Model"))
    then
        return false
    end

    return object:FindFirstChild("CurrentMap") ~= nil
        and object:FindFirstChild("RoomOwner") ~= nil
        and object:FindFirstChild("RoomOpen") ~= nil
        and object:FindFirstChild("RoomEntryMode") ~= nil
        and object:FindFirstChild("Limit") ~= nil
        and object:FindFirstChild("MapMode") ~= nil
        and object:FindFirstChild("CurrentPlayers") ~= nil
        and object:FindFirstChild("baseplate", true) ~= nil
end

function session.auto.collectElevatorRooms()
    local root = Workspace:FindFirstChild("Elevators")
    local rooms = {}
    if not root then
        return rooms
    end

    for _, group in ipairs(root:GetChildren()) do
        if group.Name ~= "ClassicElevators" then
            for _, object in ipairs(group:GetDescendants()) do
                if session.auto.isElevatorRoomCandidate(object) then
                    rooms[#rooms + 1] = object
                end
            end
        end
    end

    table.sort(rooms, function(a, b)
        local ao = a:FindFirstChild("RoomOwner")
        local bo = b:FindFirstChild("RoomOwner")
        local ar = a:FindFirstChild("RoomOpen")
        local br = b:FindFirstChild("RoomOpen")

        local aOwned = ao and tostring(ao.Value) == player.Name
        local bOwned = bo and tostring(bo.Value) == player.Name
        if aOwned ~= bOwned then
            return aOwned
        end

        local aFree = (not ao or tostring(ao.Value) == "")
            and (not ar or ar.Value == false)
        local bFree = (not bo or tostring(bo.Value) == "")
            and (not br or br.Value == false)

        if aFree ~= bFree then
            return aFree
        end

        return a:GetFullName() < b:GetFullName()
    end)

    return rooms
end

function session.auto.moveIntoElevator(room)
    local character = player.Character or player.CharacterAdded:Wait()
    local root = character:FindFirstChild("HumanoidRootPart")
        or character:FindFirstChild("Torso")
        or character.PrimaryPart
    local plate = room and room:FindFirstChild("baseplate", true)

    if not root or not plate or not plate:IsA("BasePart") then
        return false, "elevator root/baseplate missing"
    end

    local owner = room:FindFirstChild("RoomOwner")
    local roomOpen = room:FindFirstChild("RoomOpen")

    if owner
        and tostring(owner.Value) ~= ""
        and tostring(owner.Value) ~= player.Name
    then
        return false, "elevator is owned by " .. tostring(owner.Value)
    end

    if roomOpen
        and roomOpen.Value == true
        and (not owner or tostring(owner.Value) ~= player.Name)
    then
        return false, "elevator is already open"
    end

    for attempt = 1, 3 do
        if not session.alive then
            return false, "stopped"
        end

        root.CFrame = plate.CFrame * CFrame.new(0, 3, 0)

        if typeof(firetouchinterest) == "function" then
            pcall(function()
                firetouchinterest(root, plate, 0)
                task.wait(0.05)
                firetouchinterest(root, plate, 1)
            end)
        end

        local started = os.clock()
        while session.alive and os.clock() - started < 2.25 do
            if session.auto.playerInsideElevator(room) then
                return true
            end

            if owner
                and tostring(owner.Value) ~= ""
                and tostring(owner.Value) ~= player.Name
            then
                break
            end

            task.wait(0.08)
        end

        status("enter elevator retry " .. tostring(attempt + 1))
    end

    -- The native lobby can call EnterElevator on the client before RoomOwner /
    -- CurrentPlayers visibly replicate. OpenRoom is the authoritative check.
    if not owner
        or tostring(owner.Value) == ""
        or tostring(owner.Value) == player.Name
    then
        return true
    end

    return false, "EnterElevator was not confirmed"
end
