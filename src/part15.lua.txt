        allGood = allGood and good
        lines[#lines + 1] = (good and "OK  " or "MISS ") .. check[1]
    end

    local text = table.concat(lines, "\n")
    debugLog(text)
    notify(allGood and "CTDIG self-check OK" or "CTDIG self-check FAILED", text, 8)
    return allGood
end

local function cleanup()
    if not session.alive then
        return
    end

    session.alive = false
    recording = false
    replaying = false
    replayTaskRunning = false
    replayStopRequested = true

    for _, connection in ipairs(session.combatEnemySourceConnections or {}) do
        pcall(function() connection:Disconnect() end)
    end
    for _, connection in ipairs(session.combatTowerSourceConnections or {}) do
        pcall(function() connection:Disconnect() end)
    end

    for _, connection in ipairs(connections) do
        pcall(function()
            connection:Disconnect()
        end)
    end
    table.clear(connections)

    local boundTowers = {}
    for tower in pairs(towerConnections) do
        boundTowers[#boundTowers + 1] = tower
    end
    for _, tower in ipairs(boundTowers) do
        disconnectTower(tower)
    end

    session.visuals.destroyRangePreview()
    session.visuals.clearDpsGuis()
    session.visuals.clearKillHighlights()

    if Library and type(Library.Unload) == "function" then
        pcall(function()
            Library:Unload()
        end)
    end

    if environment.CTDIG_SESSION == session then
        environment.CTDIG_SESSION = nil
    end
end

session.cleanup = cleanup

local function updateDpsVisuals()
    if session.combatTowerDpsEnabled ~= true then
        session.visuals.clearDpsGuis()
        setCombatDiagDps(session.combatBuild .. " | DPS OFF | state=false")
        return
    end

    local towersFolder = getTowers()
    local alive = {}
    local now = os.clock()
    local scanned = 0
    local guiCount = 0
    local statCount = 0
    local liveCount = 0

    do
        session.bindCombatSources()

        local dpsCandidates = {}
        local staleTowers = {}

        -- Direct Workspace.Towers children are authoritative. Reconcile every
        -- visual tick so a ChildAdded replication race cannot permanently lose one.
        if session.combatTowerFolder then
            for _, tower in ipairs(session.combatTowerFolder:GetChildren()) do
                session.combatTowers[tower] = true
            end
        end

        for tower in pairs(session.combatTowers) do
            if tower
                and tower.Parent == session.combatTowerFolder
                and getRoot(tower)
            then
                dpsCandidates[#dpsCandidates + 1] = tower
            else
                staleTowers[#staleTowers + 1] = tower
            end
        end

        for _, tower in ipairs(staleTowers) do
            session.combatTowers[tower] = nil
        end

        for _, tower in ipairs(dpsCandidates) do
            if getRoot(tower) then
                scanned = scanned + 1
                alive[tower] = true

                local guiOk, data = pcall(session.visuals.ensureDpsGui, tower)
                if guiOk and data then
                    guiCount = guiCount + 1

                    local statsOk, stats = pcall(session.visuals.towerCombatStats, tower)
                    local textValue

                    if statsOk and stats then
                        statCount = statCount + 1

                        -- Stable DPS = effective damage / effective attack interval.
                        -- Never substitute bursty DeltaTotalDMG / DeltaTime here.
                        textValue = string.format(
                            "LVL %d  •  DPS %s\n%s DMG / %ss",
                            getLevel(tower),
                            session.visuals.formatCompactNumber(stats.dps),
                            session.visuals.formatCompactNumber(stats.damage),
                            session.visuals.formatCompactNumber(stats.interval)
                        )
                    else
                        local config = tower:FindFirstChild("Configuration")
                        local damage = session.visuals.readNumberValue(config, "Damage")
                        local reload = session.visuals.readNumberValue(config, "ReloadTime")
                        local fallbackDps

                        if damage and reload and reload > 0 then
                            fallbackDps = damage / reload
                        end

                        textValue = string.format(
                            "LVL %d  •  DPS %s\nDMG %s | RATE %s",
                            getLevel(tower),
                            fallbackDps and session.visuals.formatCompactNumber(fallbackDps) or "—",
                            damage ~= nil and session.visuals.formatCompactNumber(damage) or "missing",
                            reload ~= nil and session.visuals.formatCompactNumber(reload) or "missing"
                        )
                    end

                    if data.lastText ~= textValue then
                        data.lastText = textValue
                        data.label.Text = textValue
                    end
                end
            end
        end
    end

    for tower, data in pairs(dpsGuis) do
        if not alive[tower] or not tower.Parent then
            if data.gui then
                pcall(function()
                    data.gui:Destroy()
                end)
            end
            dpsGuis[tower] = nil
            dpsSamples[tower] = nil
        end
    end

    setCombatDiagDps(string.format(
        "%s | DPS towers=%d gui=%d stats=%d deterministic=true",
        session.combatBuild,
        scanned,
        guiCount,
        statCount
    ))
end
