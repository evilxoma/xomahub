    end
    for _, object in ipairs(playerGui:GetDescendants()) do
        if object.Name == "CTDIG_EnemyPredict" or object.Name == "CTDIG_DPS" then
            object:Destroy()
        end
    end
end)

    if child.Name == "Map" or child.Name == "Towers" then
        task.defer(session.bindCombatSources)
    end
end)

        or child.Name == "Towers"

    end
end)

-- Heartbeat is used instead of task.spawn for combat instance work. This avoids
-- executor thread-capability loss seen in the user's runtime.
session.lastDpsHeartbeat = 0
session.lastKillHeartbeat = 0
session.lastMaintenanceHeartbeat = 0

connect(game:GetService("RunService").Heartbeat, function()
    if not session.alive then
        return
    end

    local now = os.clock()

    if now - session.lastDpsHeartbeat >= 0.2 then
        session.lastDpsHeartbeat = now
        local dpsOk, dpsError = xpcall(updateDpsVisuals, tracebackError)
        if not dpsOk then
            combatDiagDps = session.combatBuild .. " | DPS ERROR: " .. tostring(dpsError)
            refreshCombatDiag()
            if now - lastDpsVisualErrorAt >= 5 then
                lastDpsVisualErrorAt = now
                warn("[CTDIG DPS] " .. tostring(dpsError))
            end
        end
    end

    if now - session.lastKillHeartbeat >= 0.2 then
        session.lastKillHeartbeat = now
        local killOk, killError = xpcall(session.visuals.updateKillPreview, tracebackError)
        if not killOk then
            combatDiagPredict = session.combatBuild .. " | Predict ERROR: " .. tostring(killError)
            refreshCombatDiag()
            if now - lastKillVisualErrorAt >= 5 then
                lastKillVisualErrorAt = now
                warn("[CTDIG Kill Preview] " .. tostring(killError))
            end
        end
    end

    if now - session.lastMaintenanceHeartbeat >= 0.2 then
        session.lastMaintenanceHeartbeat = now

        pcall(session.recorder.handleEndAction)

        if recording and now - lastRecorderRecoveryScan >= 0.7 then
            lastRecorderRecoveryScan = now
            pcall(function()
                session.recorder.bindGameSources()
                session.recorder.ensurePlacementRecorder()

                local towers = getTowers()
                if towers then
                    for _, tower in ipairs(towers:GetChildren()) do
                        if tower:IsA("Model") and isMine(tower) then
                            bindTower(tower)
                            if not baselineTowers[tower] and not ids[tower] then
                                session.recorder.onTowerPlaced(tower)
                            elseif ids[tower] and session.pendingTowerActions[tower] then
                                flushTowerActions(tower)
                            end
                        end
                    end
                end
            end)
        end

        if now - lastRangePreviewRefresh >= 0.25 then
            lastRangePreviewRefresh = now
            pcall(session.visuals.refreshRangePreview)
        end
    end
end)

function session.auto.runRecordedMacro()
    if not CTDIG_BUILD_AUTORUN or not session.alive then
        return
    end

    local macro, macroError = loadMacro()
    if not macro then
        setReplayStatus("AutoExecute error: " .. tostring(macroError))
        notify("AutoExecute error", macroError, 6)
        return
    end

    local started = os.clock()
    while session.alive and os.clock() - started < 30 do
        local lobbyContext = Workspace:FindFirstChild("Elevators") ~= nil
            and Workspace:FindFirstChild("Towers") == nil
        local map = Workspace:FindFirstChild("Map")
        local matchContext = Workspace:FindFirstChild("Towers") ~= nil
            and map ~= nil
            and map:FindFirstChild("Configuration") ~= nil

        if lobbyContext then
            setRecorderState("idle")
            setReplayStatus("Lobby | checking deck")
            local deckOk, deckError = session.auto.ensureMacroDeckInLobby(macro)
            if not deckOk then
                setReplayStatus("Lobby deck error: " .. tostring(deckError))
                notify("AutoExecute deck error", deckError, 8)
                return
            end

            setReplayStatus("Lobby | deck OK | finding elevator")
            local roomOk, roomError = session.auto.openMacroMapFromLobby(macro)
            if not roomOk then
                setReplayStatus("Lobby error: " .. tostring(roomError))
                notify("AutoExecute lobby error", roomError, 8)
            end
            return
        elseif matchContext then
            setReplayStatus("Match detected | waiting map data")
            local mapStarted = os.clock()
            while session.alive and os.clock() - mapStarted < 20 do
                if getMapName() ~= "Unknown" and player:FindFirstChild("PlayerData") then
                    break
                end
                task.wait(0.1)
            end

            if not session.sourceBound then
                session.sourceBound, session.sourceError = session.recorder.bindGameSources()
            end
            if not session.sourceBound then
                setReplayStatus("Match bind error: " .. tostring(session.sourceError))
                notify("AutoExecute error", session.sourceError or "match sources unavailable", 7)
                return
            end

            task.wait(0.25)
            local ok, replayError = xpcall(session.recorder.replayMacro, tracebackError)
            if not ok and session.alive then
                replaying = false
                replayTaskRunning = false
                setRecorderState("replay_stopped")
                setReplayStatus("AutoExecute runtime error")
                notify("AutoExecute replay error", tostring(replayError), 8)
            end
            return
        end

        setReplayStatus("AutoExecute | waiting for lobby/match")
        task.wait(0.25)
    end

    if session.alive then
        setReplayStatus("AutoExecute timeout: lobby/match not detected")
    end
end

local lobbyContextAtStartup = Workspace:FindFirstChild("Elevators") ~= nil
    and Workspace:FindFirstChild("Towers") == nil

if lobbyContextAtStartup then
    session.sourceBound = false
    session.sourceError = nil
    setRecorderState("idle")
    setReplayStatus(CTDIG_BUILD_AUTORUN and "AutoExecute lobby starting" or "Lobby detected")
else
    session.sourceBound, session.sourceError = session.recorder.bindGameSources()
    if session.sourceBound then
        setRecorderState("idle")
        setReplayStatus("idle")
    else
        setRecorderState("error")
        debugLog(session.sourceError)
        notify("CTDIG error", session.sourceError, 6)
    end
end

if session.saveManager then
    -- Load only after all CTDIG toggle callbacks have been registered, so a
    -- saved Auto Retry/Auto Skip/etc. immediately applies its real behavior.
    task.defer(function()
        if session.alive then
            local ok, err = pcall(function()
                session.saveManager:LoadAutoloadConfig()
            end)
            if not ok then
                warn("[CTDIG] Autoload config failed: " .. tostring(err))
            end
        end
    end)
end

if CTDIG_BUILD_AUTORUN then
    task.spawn(session.auto.runRecordedMacro)
end

-- XOMA strategy API.
-- Recorder-generated syntax stays unchanged:
-- local XOMA = loadstring(game:HttpGet(RAW_URL))()
-- local U1 = XOMA:Place(...)
-- XOMA:Upgrade(U1, ...)
-- XOMA:Run()
session.XOMA = {
    _macro = {
        version = 1,
        map = "Unknown",
        deck = {},
        config = {},
        actions = {},
    },
    _nextId = 0,
}

function session.XOMA:Map(name)
    self._macro.map = tostring(name or "Unknown")
    return self
end

function session.XOMA:Deck(deck)
    self._macro.deck = {}
    for _, unitName in ipairs(type(deck) == "table" and deck or {}) do
        if tostring(unitName or "") ~= "" and tostring(unitName) ~= "None" then
            self._macro.deck[#self._macro.deck + 1] = tostring(unitName)
        end
    end
    return self
end

function session.XOMA:Config(config)
    self._macro.config = type(config) == "table" and config or {}

    pcall(function()
        if Toggles.CTDIGAutoRetry and self._macro.config.AutoRetry ~= nil then
            Toggles.CTDIGAutoRetry:SetValue(self._macro.config.AutoRetry == true)
        end
        if Toggles.CTDIGAutoBackLobby and self._macro.config.AutoBackToLobby ~= nil then
            Toggles.CTDIGAutoBackLobby:SetValue(self._macro.config.AutoBackToLobby == true)
        end
    end)

    return self
end

function session.XOMA:_towerId(handle)
    if type(handle) == "table" then
        return handle.__xomaId or handle.id
    end
    return tonumber(handle) or handle
end

function session.XOMA:Place(unitName, position, wave, rotation)
    self._nextId = self._nextId + 1
    local id = self._nextId

    self._macro.actions[#self._macro.actions + 1] = {
        type = "place",
        id = id,
        unit = tostring(unitName or ""),
        position = vec(toVector3(position)),
        rotation = vec(toVector3(rotation or Vector3.zero)),
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }

    return {
        __xomaId = id,
        unit = tostring(unitName or ""),
    }
end

function session.XOMA:Upgrade(handle, level, wave)
    self._macro.actions[#self._macro.actions + 1] = {
        type = "upgrade",
        id = self:_towerId(handle),
        level = tonumber(level) or 1,
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }
    return handle
end

function session.XOMA:Target(handle, mode, wave)
    self._macro.actions[#self._macro.actions + 1] = {
        type = "target",
        id = self:_towerId(handle),
        mode = tostring(mode or "First"),
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }
    return handle
end

function session.XOMA:Ability(handle, ability, wave)
    self._macro.actions[#self._macro.actions + 1] = {
        type = "ability",
        id = self:_towerId(handle),
        ability = tostring(ability or "Ability1"),
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }
    return handle
end

function session.XOMA:Sell(handle, wave)
    self._macro.actions[#self._macro.actions + 1] = {
        type = "sell",
        id = self:_towerId(handle),
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }
    return handle
end

function session.XOMA:Skip(wave)
    self._macro.actions[#self._macro.actions + 1] = {
        type = "skip",
        wave = math.max(0, math.floor(tonumber(wave) or 0)),
    }
    return self
end

function session.XOMA:Run()
    local macro = self._macro
    local lobbyContext = Workspace:FindFirstChild("Elevators") ~= nil
        and Workspace:FindFirstChild("Towers") == nil

    if lobbyContext then
        local deckOk, deckError = session.auto.ensureMacroDeckInLobby(macro)
        if not deckOk then
            error("XOMA deck error: " .. tostring(deckError), 2)
        end

        local roomOk, roomError = session.auto.openMacroMapFromLobby(macro)
        if not roomOk then
            error("XOMA lobby error: " .. tostring(roomError), 2)
        end
        return true
    end

    local map = Workspace:FindFirstChild("Map")
    if not Workspace:FindFirstChild("Towers")
        or not map
        or not map:FindFirstChild("Configuration")
    then
        error("XOMA:Run() could not detect lobby or match", 2)
    end

    local ok, err = xpcall(function()
        session.recorder.replayMacro(macro)
    end, tracebackError)

    if not ok then
        error("XOMA replay error: " .. tostring(err), 2)
    end

    return true
end

return session.XOMA
